    40→type RateKey = 'bcv' | 'paralelo' | 'usdt' | 'eurBcv'
    41→
    42→/** valor actual de una tasa desde el snapshot */
    43→const snapValue = (rates: { bcv: number | null; paralelo: number | null; usdt: number | null; eurBcv: number | null } | null | undefined, key: RateKey) =>
    44→  key === 'eurBcv' ? rates?.eurBcv ?? null : rates?.[key] ?? null
    45→
    46→/** valor de una tasa en un punto del historial */
    47→const histValue = (h: RateHistoryPoint, key: RateKey) =>
    48→  key === 'eurBcv' ? h.eur : h[key]

   320→                      </p>
   321→                    </div>
   322→                  </div>
   323→                  <div className="flex items-center gap-3 mt-2.5 pt-2.5 border-t border-white/5 text-[11px] text-muted-foreground tnum flex-wrap">

   325→                      Tu tasa: <span className="text-foreground/85">{bsInvolved ? formatRate(e.impliedRate) : e.impliedRate.toFixed(3)}</span>
   326→                    </span>

   337→      </section>

   339→      <ExchangeDialog

   352→function ExchangeDialog({ open, onClose, onSaved }: { open: boolean; onClose: () => void; onSaved: () => Promise<void> }) {
   353→  const { overview } = useAppStore()
   354→  const { toast } = useToast()
   355→  const accounts = overview?.accounts ?? []
   356→  const [fromAccountId, setFromAccountId] = useState('')
   357→  const [toAccountId, setToAccountId] = useState('')
   358→  const [fromAmount, setFromAmount] = useState('')
   359→  const [toAmount, setToAmount] = useState('')
   360→  const [note, setNote] = useState('')
   361→  const [marketSource, setMarketSource] = useState('PARALELO')
   362→  const [saving, setSaving] = useState(false)
   363→
   364→  const from = accounts.find((a) => a.id === fromAccountId)
   365→  const to = accounts.find((a) => a.id === toAccountId)
   366→  const rates = overview?.rates
   367→
   368→  const implied = useMemo(() => {
   369→    const fa = parseAmount(fromAmount)
   370→    const ta = parseAmount(toAmount)
   371→    if (!from || !to || !Number.isFinite(fa) || !Number.isFinite(ta) || fa <= 0 || ta <= 0) return null
   372→    if (from.currency === 'VES' && to.currency !== 'VES') return fa / ta
   373→    if (from.currency !== 'VES' && to.currency === 'VES') return ta / fa
   374→    return ta / fa
   375→  }, [from, to, fromAmount, toAmount])
   376→
   377→  const marketRate = useMemo(() => {
   378→    if (!rates || !from || !to) return null
   379→    const usd = marketSource === 'BCV' ? rates.bcv : marketSource === 'USDT_P2P' ? rates.usdt : rates.paralelo
   380→    if (!usd) return null
   381→    const rateFor = (c: string) =>
   382→      c === 'USD' ? usd : c === 'USDT' ? (rates.usdt ?? usd) : c === 'EUR' ? usd * rates.eurUsd : 1
   383→    if (from.currency === 'VES' && to.currency !== 'VES') return rateFor(to.currency)
   384→    if (from.currency !== 'VES' && to.currency === 'VES') return rateFor(from.currency)
   385→    const rf = rateFor(from.currency)
   386→    const rt = rateFor(to.currency)
   387→    return rf && rt ? rf / rt : null
   388→  }, [rates, from, to, marketSource])
   389→
   390→  const diff = useMemo(() => {
   391→    if (implied == null || marketRate == null || !fromAmount) return null
   392→    const fa = parseAmount(fromAmount)
   393→    return (implied - marketRate) * (from?.currency === 'VES' ? parseAmount(toAmount) : fa)
   394→  }, [implied, marketRate, fromAmount, toAmount, from])
   395→
   396→  const valid =
   397→    fromAccountId && toAccountId && fromAccountId !== toAccountId && from && to && from.currency !== to.currency &&
   398→    parseAmount(fromAmount) > 0 && parseAmount(toAmount) > 0
   399→
   400→  const submit = async () => {
   401→    if (!valid || saving) return
   402→    setSaving(true)
   403→    try {
   404→      await api.createExchange({
   405→        fromAccountId,
   406→        toAccountId,
   407→        fromAmount: parseAmount(fromAmount),
   408→        toAmount: parseAmount(toAmount),
   409→        marketSource,
   410→        note: note.trim() || null,
   411→      })
   412→      toast({ title: 'Cambio registrado ✓', description: 'Los saldos de ambas cuentas se actualizaron' })
   413→      onClose()
   414→      setFromAmount('')
   415→      setToAmount('')
   416→      setNote('')
   417→      await onSaved()
   418→    } catch (e) {
   419→      toast({ title: 'Error', description: (e as Error).message, variant: 'destructive' })
   420→    } finally {
   421→      setSaving(false)
   422→    }
   423→  }
   424→
   425→  return (
   426→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
   427→      <DialogContent className="glass border-white/10 sm:max-w-md max-h-[92dvh] overflow-y-auto">
   428→        <DialogHeader>
   429→          <DialogTitle>Cambio de divisa</DialogTitle>
   430→        </DialogHeader>
   431→        <div className="space-y-4">
   432→          <div className="grid grid-cols-2 gap-3">
   433→            <div>
   434→              <Label>Entregas desde</Label>
   435→              <Select value={fromAccountId} onValueChange={setFromAccountId}>
   436→                <SelectTrigger className="bg-secondary/40 border-white/10">
   437→                  <SelectValue placeholder="Cuenta origen" />
   438→                </SelectTrigger>
   439→                <SelectContent>
   440→                  {accounts.map((a) => (
   441→                    <SelectItem key={a.id} value={a.id}>
   442→                      {a.name} ({a.currency})
   443→                    </SelectItem>
   444→                  ))}
   445→                </SelectContent>
   446→              </Select>
   447→            </div>
   448→            <div>
   449→              <Label>Recibes en</Label>
   450→              <Select value={toAccountId} onValueChange={setToAccountId}>
   451→                <SelectTrigger className="bg-secondary/40 border-white/10">
   452→                  <SelectValue placeholder="Cuenta destino" />
   453→                </SelectTrigger>
   454→                <SelectContent>
   455→                  {accounts
   456→                    .filter((a) => a.id !== fromAccountId)
   457→                    .map((a) => (
   458→                      <SelectItem key={a.id} value={a.id}>
   459→                        {a.name} ({a.currency})
   460→                      </SelectItem>
   461→                    ))}
   462→                </SelectContent>
   463→              </Select>
   464→            </div>
   465→          </div>
   466→
   467→          {from && to && from.currency === to.currency && (
   468→            <p className="text-xs text-amber-400">
   469→              Las dos cuentas usan la misma moneda: eso es una Transferencia, no un cambio.
   470→            </p>
   471→          )}
   472→
   473→          <div className="grid grid-cols-2 gap-3">
   474→            <div>

   531→            {saving ? 'Registrando…' : 'Registrar cambio'}
   532→          </Button>
   533→        </DialogFooter>
   534→      </DialogContent>
   535→    </Dialog>
   536→  )

   539→function ManualRateDialog({ open, onClose }: { open: boolean; onClose: () => void }) {
   540→  const { refreshOverview } = useAppStore()
   541→  const { toast } = useToast()
   542→  const [source, setSource] = useState('PARALELO')
   543→  const [usd, setUsd] = useState('')
   544→  const [eur, setEur] = useState('')
   545→  const [saving, setSaving] = useState(false)
   546→
   547→  const submit = async () => {
   548→    const v = parseAmount(usd)
   549→    const ev = parseAmount(eur)
   550→    if (!Number.isFinite(v) || v <= 0 || saving) return
   551→    setSaving(true)
   552→    try {
   553→      await api.setRate(source, v, Number.isFinite(ev) && ev > 0 ? ev : undefined)
   554→      toast({ title: 'Tasa fijada manualmente ✓' })
   555→      onClose()
   556→      setUsd('')
   557→      setEur('')
   558→      await refreshOverview()
   559→    } catch (e) {
   560→      toast({ title: 'Error', description: (e as Error).message, variant: 'destructive' })
   561→    } finally {
   562→      setSaving(false)
   563→    }
   564→  }
   565→
   566→  return (
   567→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
